const storageKey = 'cartItems';

const saveCart = async items => {
  await AsyncStorage.setItem(storageKey, JSON.stringify(items));

const useCartStore = create(set => ({
  items: __DEV__ && devFill ? devCart : [],
  addItem: item =>
    set(state => {
      const existingItem = state.items.find(i => i.url === item.url);
      if (existingItem) {
        return {
          items: state.items.map(i =>
            i.url === item.url ? {...i, count: i.count + 1} : i,
          ),
        };
      } else {
        return {items: [...state.items, {...item, count: 1}]};
      }
    }),
  removeItem: url =>
    set(state => ({
      items: state.items.filter(i => i.url !== url),
    })),
  increaseCount: url =>
    set(state => ({
      items: state.items.map(i =>
        i.url === url ? {...i, count: i.count + 1} : i,
      ),
    })),
  decreaseCount: url =>
    set(state => ({
      items: state.items
        .map(i =>
          i.url === url && i.count > 1 ? {...i, count: i.count - 1} : i,
        )
        .filter(i => i.count > 0),
    })),
  updateItem: (url, item) =>
    set(state => ({
      items: state.items.map(i => (i.url === url ? {...i, ...item} : i)),
    })),
  updateField: (url, field, value) =>
    set(state => ({
      items: state.items.map(i => (i.url === url ? {...i, [field]: value} : i)),
    })),
  clearCart: () => {
    return set({items: []})
  },
  loadCart: async () => {
    const storedItems = await AsyncStorage.getItem(storageKey);
    console.log('storedItems', storedItems);
    if (storedItems) {
      set({items: JSON.parse(storedItems)});
    }
  },



}));
